class Solution {
public:
    int findMaxConsecutiveOnes(vector<int>& nums) {
        int n = nums.size();
        int cnt = 0;
        bool start = true;
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (nums[i] == 1) {
                if (start) {
                    cnt = 1;
                    start = false;
                } else {
                    cnt++;
                }
                if (i == n - 1) {
                    ans = max(ans, cnt);
                    start = true;
                }
            } else {
                ans = max(ans, cnt);
                start = true;
            }
        }
        return ans;
    }
};

Explanation : use start to decide if the current index is the start of a new subsequence of 1's and cnt to find the length of each subsequence ,then store the max length in ans and return it

Problem 2 : Longest Substring Without Repeating Characters

class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int l = 0, r = 1;
        bool visited[300] = {false};
        int n = s.ngth();

        if (n == 0)
            return 0;
        if (n == 1)
            return 1;

        int ans = 0, len = 0;
        visited[s[l] - 0] = true;
        len++;
        ans = len;

        while (r < n) {
            if (l == r) {
                len = 1;
                visited[s[l] - 0] = true;
                r++;
            } else if (!visited[s[r] - 0]) {
                len++;
                visited[s[r] - 0] = true;
                if (ans < len)
                    ans = len;

                r++;
            } else {
                while (s[l] != s[r]) {
                    visited[s[l] - 0] = false;
                    l++;
                }
                visited[s[l] - 0] = false;

                l++;
                len = r - l;
            }
        }

        return ans;
    }
};

Explanation : use a boolean visited array to keep track of Characters present in the current window ,
                if current window size is 1 mark the current Character as visited and then if it is not 1 if 
                the current Character is not visited yet add 1 to the window size else shorten the window to include the Character just after the previous appearance of the current Character

Problem 3 : Sliding Window Maximum

class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int n = nums.size();
        vector<int>ans;
        vector<int>prefix_max(n);
        vector<int>suffix_max(n);
        for(int i =0;i<n;i++){
            if(i%k==0){
                prefix_max[i] = nums[i];
            }else{
                prefix_max[i] = max(prefix_max[i-1],nums[i]);
            }
        }
        for(int i=n-1;i>=0;--i){
            if(i==n-1 || (i+1)%k==0){
                suffix_max[i]=nums[i];
            }else{
                suffix_max[i] = max(suffix_max[i+1],nums[i]);
            }
        }
        for(int i=0;i<=n-k;i++){
            int M = max(suffix_max[i],prefix_max[i+k-1]);
            ans.push_back(M);
        }
        return ans;
    }
};

Explanation : divide nums into independent blocks of size k and calculate prefix_max and suffix_max for these
            then for any window of size k starting from i to i+k-1 
            if i%k ==0 it coincides with one of our independent blocks so the answer would be 
                prefix_max[i+k-1] or suffix_max[i]; both are equal
            else if not then this current window overlaps with 2 adjacent blocks so then the ans would be 
                max(prefix_max[i+k-1],suffix_max[i])
            so finally the max value in the window is 
                max(prefix_max[i+k-1],suffix_max[i])
                no matter what i is

Problem 4 : Minimum Window Substring

class Solution {
public:
    string minWindow(string s, string t) {
        if (s.empty() || t.empty() || s.length() < t.length()) {
            return "";
        }

        vector<int> freq(128, 0);
        for (char c : t) {
            freq[c]++;
        }

        int left = 0, right = 0;
        int minStart = 0, minLen = INT_MAX;

        int counter = t.length();

        while (right < s.length()) {
            if (freq[s[right]] > 0) {
                counter--;
            }
            freq[s[right]]--;
            right++;

            while (counter == 0) {

                if (right - left < minLen) {
                    minStart = left;
                    minLen = right - left;
                }

                freq[s[left]]++;

                if (freq[s[left]] > 0) {
                    counter++;
                }
                left++;
            }
        }

        return minLen == INT_MAX ? "" : s.substr(minStart, minLen);
    }
};

Explanation: Use two pointers (left and right) and a frequency array to keep track of the characters required from string t. Expand the window by moving right and decreasing a counter whenever a required character is included. When the counter reaches 0, it means all necessary characters are in the window; at this point, update the minimum window length (minLen) and try shrinking the window from the left to find the smallest valid substring. Finally, extract and return the substring using the optimal start index and length

Problem 7 : Minimum size subarray sum


class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
       int n = nums.size();
       int ans  = 0;
       long long sum = 0;
       for(auto x:nums)sum+=x;
       if(sum<target)return ans;
        ans = n;
        vector<long long >prefix(n);
        prefix[0] = nums[0];
        for(int i =1;i<n;i++){
            prefix[i] = prefix[i-1]+nums[i];
        }
        auto it = lower_bound(prefix.begin(),prefix.end(),target);
        int idx = distance(prefix.begin(),it);
        ans = min(ans,idx+1);
        for(int i = 0;i< n;i++){
            it = lower_bound(prefix.begin(),prefix.end(),target+prefix[i]);
            if(it == prefix.end())break;
            idx = distance(prefix.begin(),it);
            ans = min(ans,idx-i);
        }
        return ans;
    }
};

Explanation: First, compute a prefix sum array to quickly calculate subarray sums. Iterate through the array and use binary search (lower_bound) to find the smallest ending index where the sum is greater than or equal to the target plus the prefix sum up to the current starting point i. Calculate the length of this valid subarray and continuously update the minimum length found


Problem 8 : Sliding window mode

 
#include <bits/stdc++.h>
using namespace std;

#define fastio()                                                               \
  ios_base::sync_with_stdio(false);                                            \
  cin.tie(NULL);                                                               \
  cout.tie(NULL)
#define ll long long
#define nline "\n"

void solve() {
  int n, k;
  cin >> n >> k;
  vector<ll> a(n);
  for (auto &x : a)
    cin >> x;

  map<ll, int> freq;
  map<int, set<ll>> inv_freq;

  for (int i = 0; i < n; i++) {
    int old_f = freq[a[i]];
    if (old_f > 0) {
      inv_freq[old_f].erase(a[i]);
      if (inv_freq[old_f].empty()) {
        inv_freq.erase(old_f);
      }
    }
    freq[a[i]]++;
    inv_freq[freq[a[i]]].insert(a[i]);

    if (i >= k) {
      ll drop_val = a[i - k];
      int current_f = freq[drop_val];

      inv_freq[current_f].erase(drop_val);
      if (inv_freq[current_f].empty()) {
        inv_freq.erase(current_f);
      }

      freq[drop_val]--;
      if (freq[drop_val] > 0) {
        inv_freq[freq[drop_val]].insert(drop_val);
      } else {
        freq.erase(drop_val);
      }
    }

    if (i >= k - 1) {
      auto mode = inv_freq.rbegin();
      auto &m = mode->second;
      cout << *m.begin() << " ";
    }
  }
  cout << nline;
}

int main() {
  fastio();
  int t = 1;
  while (t--) {
    solve();
  }
  return 0;
}

Explanation: Maintain two maps: one for tracking the frequency of elements (freq) and another mapping frequencies back to a sorted set of elements (inv_freq). As the window slides, increment the frequency of the new element and decrement the frequency of the element dropping out (a[i - k]), updating both maps accordingly. The mode is efficiently retrieved by taking the highest frequency from inv_freq using rbegin() and picking the smallest element from its set

  vector<ll> nums(n);
  nums[0] = x;
  for (int i = 1; i < n; i++) {
    nums[i] = (a * nums[i - 1] + b) % c;

  for (int i = 0; i < n; i++) {

For each window of size k, calculate its total XOR by XORing the current prefix with the prefix just before the window starts (prefix[i] ^ prefix[i - k]), and accumulate these window XORs to get the final answer


Problem 10 : 3Sum

class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>> ans;
        sort(nums.begin(), nums.end());
        for (int i = 0; i < n; i++) {
            if (i > 0 && nums[i] == nums[i - 1])
                continue;
            int l = i + 1, r = n - 1;
            while (l < r) {
                if (nums[l] + nums[r] + nums[i] == 0) {
                    ans.push_back({nums[i], nums[l], nums[r]});
                    ++l;
                    --r;
                    while (l < r && nums[l] == nums[l - 1])
                        l++;
                    while (l < r && nums[r] == nums[r + 1])
                        r--;
                } else if (nums[l] + nums[r] + nums[i] > 0) {
                    --r;
                } else {
                    ++l;
                }
            }
        }
        return ans;
    }
};

Explanation: Sort the array to easily manage duplicate values and utilize a two-pointer approach. Fix one number at index i, and use two pointers (l and r) to find pairs that sum to -nums[i]. When a valid triplet is found, push it to the answer and carefully skip past duplicate values for both pointers to ensure all recorded triplets are unique

        for (int i = 0; i < n; i++) {
            if (region[s[i] - 'a'].first == -1) {

            } else {
                region[s[i] - 'a'].second = i;
            }

        int l, r;
        bool start = true;
        int cnt = 0;

            if (start) {
                l = currRegionStart;
                r = currRegionEnd;
                start = false;
                
            } else {

                    r = max(r, currRegionEnd);
                } else {
                    ans.push_back(cnt);
                    l = currRegionStart;
                    r = currRegionEnd;
                }
            }
            cnt = r-l+1;
            if(i==25){
                ans.push_back(cnt);
            }
        }
        return ans;
    }

Problem 13 : Trapping rain water

class Solution {
public:
    int trap(vector<int>& height) {
        int n = height.size();
        if (n <= 2)
            return 0;
        int left = 0, right = n - 1;
        int left_max = 0, right_max = 0;
        int total_water = 0;

        while (left < right) {
            if (height[left] < height[right]) {
                if (height[left] >= left_max) {
                    left_max = height[left];
                } else {
                    total_water += left_max - height[left];
                }
                left++;
            } else {
                if (height[right] >= right_max) {
                    right_max = height[right];
                } else {
                    total_water += right_max - height[right];
                }
                right--;
            }
        }

        return total_water;
    }
};

Explanation: Utilize two pointers (left and right) along with variables tracking the maximum heights seen from both the left and right sides. Compare the heights at the two pointers: process the smaller side by taking the difference between its maximum and its current height, adding that difference to the total water, and moving the respective pointer inward

    int pivotIndex(vector<int>& nums) {
        int n = nums.size();
        vector<int>prefix(n);
        for(int i=0;i<n;i++){
            if(i==0){
                prefix[i]=nums[i];
            }else{
                prefix[i]=prefix[i-1]+nums[i];
            }
        }

class Solution {

    string shiftingLetters(string s, vector<int>& shifts) {
        int n = s.length();
        vector<long long>suffix(n);
        for (int i = n - 1; i >= 0; i--) {
            if (i == n - 1) {
                suffix[i] = shifts[i];

                suffix[i] = suffix[i + 1] + shifts[i];
            }
        }

            int currChar = s[i]-'a';

            char ch = 'a' + currChar;
            ans+=ch;
        }
        return ans;
    }
};

Explanation: Since each shift operation affects the current character and all subsequent characters, compute a suffix sum of the shifts array. Problem 8 : XOR queries of a subarray

class Solution {
public:
    vector<int> xorQueries(vector<int>& arr, vector<vector<int>>& queries) {
        int n = arr.size();
        int q = queries.size();
        vector<long long> prefix(n);
        for (int i = 0; i < n; i++) {
            if (i == 0) {
                prefix[i] = arr[i];
            } else {
                prefix[i] = prefix[i - 1] ^ arr[i];
            }
        }
        vector<int> ans(q);
        for (int i = 0; i < q; i++) {
            int left = queries[i][0];
            int right = queries[i][1];
            if (left > 0) {
                ans[i] = prefix[right] ^ prefix[left - 1];
            } else {
                ans[i] = prefix[right];
            }
        }
        return ans;
    }
};

Explanation: Precompute a prefix XOR array where each element stores the XOR sum of all previous elements up to that index. To process a query [left, right], use the cancellation property of XOR: the result is prefix[right] ^ prefix[left - 1] (if left > 0), which effectively removes the values before the left boundary

#define mp make_pair
#define nline "\n"

void solve() {
  int n;
  cin >> n;
  vector<int> a(n);
  for (auto &x : a)
    cin >> x;
  map<int, int> freq;
  for (int i = 0; i < n; i++) {
    freq[a[i] - i]++;
  }
  ll ans = 0;
  for (auto &[key, val] : freq) {
    ans += (ll)val * (val - 1) / 2;
  }
  cout << ans << nline;
}

int main() {
  fastio();
  int t;
  cin >> t;
  while (t--) {
    solve();
  }
  return 0;
}

Explanation: The condition a[j] - a[i] = j - i can be algebraically rewritten as a[j] - j = a[i] - i. Create a frequency map to count how many times each value of a[i] - i appears. For every unique value that appears val times, it contributes val * (val - 1) / 2 valid pairs to the final count
